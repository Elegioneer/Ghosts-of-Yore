				has_faction_template = faction_template_goy_pact_of_bronze_faction

			}
		}
	}
	available = {
		
	}
	completed = {
		count_in_collection = {

		    }
			unit = carrier

		}
		count_in_collection = {
			collection = {
				input = game:scope

				name = FACTION
		    }

		}
